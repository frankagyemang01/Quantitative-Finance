Daily stock returns are close to noise (efficient markets), so classical “seasonality” is usually negligible – this is included because it was requested, but interpret weak seasonal/cyclical components with caution. STL (stl()) splits a series into trend + seasonal + remainder only. It does NOT separate a distinct “cyclical” component – cycles longer than the seasonal period end up folded into the trend. To pull out a genuine business-cycle-length component, we additionally apply an HP (Hodrick-Prescott) filter to the STL trend, which splits it into a smooth long-run trend and a cyclical deviation from it.

##  Decomposition: trend, seasonality, cyclicality, residual  
##   This requires the `mFilter` package

decompose_stock <- function(ret_xts, ticker, frequency = 21) {
  # frequency = 21 treats a trading month as the "seasonal" period;
  # use 5 for a trading week, or 252 for an annual cycle instead.
  ts_data <- ts(as.numeric(ret_xts), frequency = frequency)

  stl_fit <- stl(ts_data, s.window = "periodic", robust = TRUE)
  comp <- as.data.frame(stl_fit$time.series)  # seasonal, trend, remainder 
  # Further split the STL trend into long-run trend + cycle via HP filter
  hp_fit <- hpfilter(comp$trend, freq = 14400)  # 14400 is a common daily-data lambda
  comp$long_run_trend <- as.numeric(hp_fit$trend)
  comp$cycle          <- as.numeric(hp_fit$cycle)

  comp$Date   <- index(ret_xts)
  comp$Ticker <- ticker
  comp
}
decomp_list <- lapply(tickers, function(t) decompose_stock(returns_log [, t], t))
decomp_all  <- bind_rows(decomp_list)
##  Plot each stock's decomposition 
plot_decomposition <- function(df, ticker) {
  df_long <- df %>%
    filter(Ticker == ticker) %>%
    select(Date, seasonal, long_run_trend, cycle, remainder) %>%
    rename(Trend = long_run_trend, Seasonal = seasonal,
           Cyclical = cycle, Residual = remainder) %>%
    pivot_longer(-Date, names_to = "Component", values_to = "Value")

  ggplot(df_long, aes(x = Date, y = Value)) +
    geom_line(color = "steelblue", linewidth = 0.4) +
    facet_wrap(~ Component, scales = "free_y", ncol = 1) +
    labs(title = paste("Decomposition of Daily Returns:", ticker),
         x = "Date", y = NULL) +
    theme_minimal()
}

# Example: plot decomposition for each ticker
for (t in tickers) {
  print(plot_decomposition(decomp_all, t))
}
